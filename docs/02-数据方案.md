## 二、气象数据详解

### 2.1 风场数据 (Wind)

**数据源**：NOAA GFS
- URL: https://nomads.ncep.noaa.gov/
- 格式：GRIB2
- 分辨率：0.25° (约 25km)
- 覆盖：全球
- 参数：
  - UGRD (U 分量，东西向)
  - VGRD (V 分量，南北向)
  - 高度：10m

**下载示例**：
```bash
# 下载最新的 GFS 数据
wget "https://nomads.ncep.noaa.gov/cgi-bin/filter_gfs_0p25.pl?\
file=gfs.t00z.pgrb2.0p25.f000&\
lev_10_m_above_ground=on&\
var_UGRD=on&var_VGRD=on&\
dir=%2Fgfs.20240101%2F00%2Fatmos"
```

**原始格式 (GRIB2)**：
```
二进制格式，无法直接读取
需要使用 grib2json 转换
```

**转换后格式 (JSON)**：
```json
[
  {
    "header": {
      "discipline": 0,
      "parameterCategory": 2,
      "parameterNumber": 2,
      "refTime": "2024-01-01T00:00:00.000Z",
      "forecastTime": 0,
      "surface1Type": 103,
      "surface1Value": 10.0,
      "gridDefinitionTemplate": 0,
      "nx": 1440,
      "ny": 721,
      "lo1": 0,
      "la1": 90,
      "lo2": 359.75,
      "la2": -90,
      "dx": 0.25,
      "dy": 0.25
    },
    "data": [
      5.2, -3.1, 2.8, ...  // U 分量数组
    ]
  },
  {
    "header": {...},
    "data": [...]  // V 分量数组
  }
]
```

**我们需要的格式**：
```json
{
  "refTime": "2024-01-01T00:00:00Z",
  "forecastTime": 0,
  "bounds": {
    "west": 0,
    "south": -90,
    "east": 360,
    "north": 90
  },
  "resolution": 0.25,
  "data": [
    {"u": 5.2, "v": -3.1},  // 第一个点
    {"u": 2.8, "v": 1.5},   // 第二个点
    ...
  ]
}
```

### 2.2 波浪数据 (Wave)

**数据源**：NOAA WaveWatch III
- URL: https://polar.ncep.noaa.gov/waves/
- 格式：GRIB2
- 分辨率：0.5°
- 参数：
  - HTSGW (有效波高)
  - PERPW (波浪周期)
  - DIRPW (波浪方向)

**我们需要的格式**：
```json
{
  "refTime": "2024-01-01T00:00:00Z",
  "data": [
    {
      "height": 2.5,      // 波高 (米)
      "period": 8.0,      // 周期 (秒)
      "direction": 270    // 方向 (度，0=北)
    },
    ...
  ]
}
```

### 2.3 洋流数据 (Current)

**数据源**：HYCOM
- URL: https://www.hycom.org/dataserver/gofs-3pt1/analysis
- 格式：NetCDF
- 分辨率：1/12° (约 8km)
- 参数：
  - water_u (东西向流速)
  - water_v (南北向流速)
  - 深度：0m (表层)

**我们需要的格式**：
```json
{
  "refTime": "2024-01-01T00:00:00Z",
  "data": [
    {
      "u": 0.5,    // 东西向流速 (m/s)
      "v": -0.3    // 南北向流速 (m/s)
    },
    ...
  ]
}
```

## 四、数据处理流程

### 4.1 气象数据处理

```
1. 下载 GRIB2 文件
   ↓
2. 使用 grib2json 转换为 JSON
   ↓
3. 解析 JSON，提取需要的字段
   ↓
4. 转换为统一格式
   ↓
5. 压缩 (gzip)
   ↓
6. 存储到文件系统
   ↓
7. 缓存到 Redis (30 分钟)
```

**代码示例**：
```javascript
// 1. 下载
const axios = require('axios');
const fs = require('fs');

async function downloadGFS() {
  const url = 'https://nomads.ncep.noaa.gov/...';
  const response = await axios.get(url, {
    responseType: 'arraybuffer'
  });
  fs.writeFileSync('gfs.grib2', response.data);
}

// 2. 转换
const { exec } = require('child_process');

function convertGRIB2() {
  return new Promise((resolve, reject) => {
    exec('grib2json -d -n -o gfs.json gfs.grib2', (error) => {
      if (error) reject(error);
      else resolve();
    });
  });
}

// 3. 解析
function parseGFS() {
  const raw = JSON.parse(fs.readFileSync('gfs.json'));
  
  const uData = raw.find(r => r.header.parameterNumber === 2);
  const vData = raw.find(r => r.header.parameterNumber === 3);
  
  const result = {
    refTime: uData.header.refTime,
    data: []
  };
  
  for (let i = 0; i < uData.data.length; i++) {
    result.data.push({
      u: uData.data[i],
      v: vData.data[i]
    });
  }
  
  return result;
}
```

### 4.2 航线气象匹配

```
1. 获取航线 waypoints
   ↓
2. 对每个 waypoint，计算其在气象网格中的位置
   ↓
3. 使用双线性插值获取该点的气象数据
   ↓
4. 返回航线上所有点的气象数据
```

**代码示例**：
```javascript
function getWeatherAlongRoute(route, weatherData) {
  const result = [];
  
  for (const waypoint of route.waypoints) {
    const weather = interpolateWeather(
      waypoint.lat,
      waypoint.lon,
      weatherData
    );
    
    result.push({
      ...waypoint,
      wind: {
        speed: Math.sqrt(weather.u ** 2 + weather.v ** 2),
        direction: Math.atan2(weather.u, weather.v) * 180 / Math.PI
      }
    });
  }
  
  return result;
}

// 双线性插值
function interpolateWeather(lat, lon, weatherData) {
  const { bounds, resolution, data } = weatherData;
  
  // 计算网格索引
  const x = (lon - bounds.west) / resolution;
  const y = (bounds.north - lat) / resolution;
  
  const x0 = Math.floor(x);
  const y0 = Math.floor(y);
  const x1 = x0 + 1;
  const y1 = y0 + 1;
  
  // 获取四个角的数据
  const q00 = data[y0 * nx + x0];
  const q10 = data[y0 * nx + x1];
  const q01 = data[y1 * nx + x0];
  const q11 = data[y1 * nx + x1];
  
  // 双线性插值
  const wx = x - x0;
  const wy = y - y0;
  
  const u = (1 - wx) * (1 - wy) * q00.u +
            wx * (1 - wy) * q10.u +
            (1 - wx) * wy * q01.u +
            wx * wy * q11.u;
            
  const v = (1 - wx) * (1 - wy) * q00.v +
            wx * (1 - wy) * q10.v +
            (1 - wx) * wy * q01.v +
            wx * wy * q11.v;
  
  return { u, v };
}
```

## 五、数据存储方案

### 5.1 文件存储

```
/data
  /weather
    /wind
      2024-01-01T00.json.gz
      2024-01-01T06.json.gz
      ...
    /wave
      2024-01-01T00.json.gz
      ...
    /current
      2024-01-01T00.json.gz
      ...
```

### 5.2 数据库存储

**PostgreSQL 表结构**：
```sql
-- 航线表
CREATE TABLE routes (
  id SERIAL PRIMARY KEY,
  mmsi BIGINT NOT NULL,
  vessel_name VARCHAR(255),
  departure_port VARCHAR(255),
  departure_time TIMESTAMP,
  destination_port VARCHAR(255),
  eta TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW()
);

-- 航点表
CREATE TABLE waypoints (
  id SERIAL PRIMARY KEY,
  route_id INTEGER REFERENCES routes(id),
  lat DOUBLE PRECISION,
  lon DOUBLE PRECISION,
  time TIMESTAMP,
  speed DOUBLE PRECISION,
  course DOUBLE PRECISION,
  wind_speed DOUBLE PRECISION,
  wind_direction DOUBLE PRECISION,
  wave_height DOUBLE PRECISION,
  current_speed DOUBLE PRECISION
);

-- 创建空间索引
CREATE INDEX idx_waypoints_location 
ON waypoints USING GIST (ST_MakePoint(lon, lat));
```

### 5.3 Redis 缓存

```
# 气象数据缓存 (30 分钟)
weather:wind:2024-01-01T00 -> JSON
weather:wave:2024-01-01T00 -> JSON
weather:current:2024-01-01T00 -> JSON

# 航线缓存 (10 分钟)
route:413123456 -> JSON

# 航线气象缓存 (10 分钟)
route:413123456:weather -> JSON
```

## 六、数据更新策略

```javascript
// 使用 node-cron 定时任务
const cron = require('node-cron');

// 每 6 小时更新一次
cron.schedule('0 */6 * * *', async () => {
  console.log('开始更新气象数据...');
  
  await downloadGFS();
  await convertGRIB2();
  const windData = parseGFS();
  await saveWeatherData('wind', windData);
  
  console.log('气象数据更新完成');
});
```
