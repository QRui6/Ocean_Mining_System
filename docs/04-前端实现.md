### 2.1 风场粒子系统

**原理**：使用 Cesium PointPrimitiveCollection 渲染粒子

```javascript
// utils/weatherRenderer.js
import * as Cesium from 'cesium';

export class WindParticleSystem {
  constructor(viewer, options = {}) {
    this.viewer = viewer;
    this.particleCount = options.particleCount || 5000;
    this.speedFactor = options.speedFactor || 0.5;
    this.particles = [];
    this.collection = null;
    this.windData = null;
    this.isActive = false;
  }
  
  // 加载风场数据
  async loadWindData(windData) {
    this.windData = windData;
    console.log('✅ 风场数据已加载');
  }
  
  // 启动粒子系统
  start() {
    if (!this.windData) {
      console.error('❌ 请先加载风场数据');
      return;
    }
    
    this.isActive = true;
    this.initParticles();
    this.animate();
    
    console.log('🌬️ 风场粒子系统已启动');
  }
  
  // 初始化粒子
  initParticles() {
    // 创建粒子集合
    this.collection = this.viewer.scene.primitives.add(
      new Cesium.PointPrimitiveCollection()
    );
    
    // 创建粒子
    for (let i = 0; i < this.particleCount; i++) {
      const lon = Math.random() * 360 - 180;
      const lat = Math.random() * 180 - 90;
      const age = Math.random() * 100;
      
      const particle = {
        lon,
        lat,
        age,
        maxAge: 100,
        speed: { u: 0, v: 0 }
      };
      
      this.particles.push(particle);
      
      // 添加到渲染集合
      this.collection.add({
        position: Cesium.Cartesian3.fromDegrees(lon, lat, 100000),
        color: Cesium.Color.CYAN.withAlpha(0.6),
        pixelSize: 2
      });
    }
  }
  
  // 动画循环
  animate() {
    if (!this.isActive) return;
    
    this.particles.forEach((particle, i) => {
      // 获取当前位置的风速
      const wind = this.getWindAt(particle.lon, particle.lat);
      
      if (wind) {
        // 更新位置
        particle.lon += wind.u * this.speedFactor * 0.01;
        particle.lat += wind.v * this.speedFactor * 0.01;
        
        // 边界处理
        if (particle.lon > 180) particle.lon -= 360;
        if (particle.lon < -180) particle.lon += 360;
        if (particle.lat > 90 || particle.lat < -90) {
          this.resetParticle(particle);
        }
        
        // 更新年龄
        particle.age += 1;
        if (particle.age > particle.maxAge) {
          this.resetParticle(particle);
        }
        
        // 更新渲染位置
        const point = this.collection.get(i);
        if (point) {
          point.position = Cesium.Cartesian3.fromDegrees(
            particle.lon,
            particle.lat,
            100000
          );
          
          // 根据风速调整颜色
          const speed = Math.sqrt(wind.u ** 2 + wind.v ** 2);
          const alpha = Math.min(speed / 20, 1) * (1 - particle.age / particle.maxAge);
          point.color = Cesium.Color.CYAN.withAlpha(alpha);
        }
      }
    });
    
    requestAnimationFrame(() => this.animate());
  }
  
  // 获取指定位置的风速
  getWindAt(lon, lat) {
    if (!this.windData) return null;
    
    const { bounds, resolution, nx, ny, data } = this.windData;
    
    // 计算网格索引
    const x = Math.floor((lon - bounds.west) / resolution);
    const y = Math.floor((bounds.north - lat) / resolution);
    
    if (x < 0 || x >= nx || y < 0 || y >= ny) return null;
    
    const index = y * nx + x;
    return data[index] || null;
  }
  
  // 重置粒子
  resetParticle(particle) {
    particle.lon = Math.random() * 360 - 180;
    particle.lat = Math.random() * 180 - 90;
    particle.age = 0;
  }
  
  // 停止粒子系统
  stop() {
    this.isActive = false;
    if (this.collection) {
      this.viewer.scene.primitives.remove(this.collection);
      this.collection = null;
    }
    this.particles = [];
    console.log('⏹️ 风场粒子系统已停止');
  }
}
```

### 2.2 气象控制面板

```vue
<!-- components/WeatherControl.vue -->
<template>
  <div class="weather-control">
    <div class="panel-header">
      <h3>气象图层</h3>
    </div>
    
    <div class="panel-content">
      <!-- 风场控制 -->
      <div class="layer-item">
        <label>
          <input type="checkbox" v-model="layers.wind" @change="toggleWind">
          <span>🌬️ 风场</span>
        </label>
        <div v-if="layers.wind" class="layer-options">
          <label>
            粒子数量: {{ windParticleCount }}
            <input type="range" v-model="windParticleCount" 
                   min="1000" max="10000" step="1000"
                   @change="updateWindParticles">
          </label>
        </div>
      </div>
      
      <!-- 波浪控制 -->
      <div class="layer-item">
        <label>
          <input type="checkbox" v-model="layers.wave" @change="toggleWave">
          <span>🌊 波浪</span>
        </label>
      </div>
      
      <!-- 洋流控制 -->
      <div class="layer-item">
        <label>
          <input type="checkbox" v-model="layers.current" @change="toggleCurrent">
          <span>🌀 洋流</span>
        </label>
      </div>
    </div>
  </div>
</template>

<script>
import { ref } from 'vue';
import { useWeatherStore } from '@/stores/weatherStore';
import { WindParticleSystem } from '@/utils/weatherRenderer';

export default {
  props: {
    viewer: Object
  },
  setup(props) {
    const weatherStore = useWeatherStore();
    
    const layers = ref({
      wind: false,
      wave: false,
      current: false
    });
    
    const windParticleCount = ref(5000);
    let windSystem = null;
    
    // 切换风场
    const toggleWind = async () => {
      if (layers.value.wind) {
        // 启动
        windSystem = new WindParticleSystem(props.viewer, {
          particleCount: windParticleCount.value
        });
        
        const windData = await weatherStore.getWindData();
        await windSystem.loadWindData(windData);
        windSystem.start();
      } else {
        // 停止
        if (windSystem) {
          windSystem.stop();
          windSystem = null;
        }
      }
    };
    
    // 更新粒子数量
    const updateWindParticles = () => {
      if (windSystem) {
        windSystem.stop();
        toggleWind();
      }
    };
    
    return {
      layers,
      windParticleCount,
      toggleWind,
      toggleWave: () => {},
      toggleCurrent: () => {},
      updateWindParticles
    };
  }
};
</script>
```

### 2.3 时序播放器

```vue
<!-- components/TimelinePlayer.vue -->
<template>
  <div class="timeline-player">
    <div class="timeline-header">
      <span>{{ currentTime }}</span>
    </div>
    
    <div class="timeline-controls">
      <button @click="playPause">
        {{ isPlaying ? '⏸️' : '▶️' }}
      </button>
      <button @click="stop">⏹️</button>
      <button @click="stepBackward">⏮️</button>
      <button @click="stepForward">⏭️</button>
    </div>
    
    <div class="timeline-slider">
      <input type="range" 
             v-model="currentIndex" 
             :min="0" 
             :max="timeSteps.length - 1"
             @input="onSliderChange">
      <div class="timeline-labels">
        <span>{{ startTime }}</span>
        <span>{{ endTime }}</span>
      </div>
    </div>
    
    <div class="timeline-speed">
      <label>
        播放速度: {{ playSpeed }}x
        <input type="range" v-model="playSpeed" 
               min="0.5" max="4" step="0.5">
      </label>
    </div>
  </div>
</template>

<script>
import { ref, computed, watch } from 'vue';
import { useWeatherStore } from '@/stores/weatherStore';

export default {
  setup() {
    const weatherStore = useWeatherStore();
    
    const isPlaying = ref(false);
    const currentIndex = ref(0);
    const playSpeed = ref(1);
    const timeSteps = ref([]);
    
    let animationId = null;
    let lastUpdateTime = 0;
    
    // 当前时间
    const currentTime = computed(() => {
      if (timeSteps.value.length === 0) return '';
      return new Date(timeSteps.value[currentIndex.value]).toLocaleString();
    });
    
    // 开始/结束时间
    const startTime = computed(() => {
      if (timeSteps.value.length === 0) return '';
      return new Date(timeSteps.value[0]).toLocaleDateString();
    });
    
    const endTime = computed(() => {
      if (timeSteps.value.length === 0) return '';
      return new Date(timeSteps.value[timeSteps.value.length - 1]).toLocaleDateString();
    });
    
    // 加载时间序列
    const loadTimeline = async () => {
      const timeline = await weatherStore.getTimeline();
      timeSteps.value = timeline.map(t => t.time);
    };
    
    // 播放/暂停
    const playPause = () => {
      isPlaying.value = !isPlaying.value;
      
      if (isPlaying.value) {
        animate();
      } else {
        if (animationId) {
          cancelAnimationFrame(animationId);
          animationId = null;
        }
      }
    };
    
    // 动画循环
    const animate = (timestamp = 0) => {
      if (!isPlaying.value) return;
      
      const elapsed = timestamp - lastUpdateTime;
      const interval = 1000 / playSpeed.value;  // 根据速度调整间隔
      
      if (elapsed > interval) {
        currentIndex.value++;
        
        if (currentIndex.value >= timeSteps.value.length) {
          currentIndex.value = 0;  // 循环播放
        }
        
        lastUpdateTime = timestamp;
      }
      
      animationId = requestAnimationFrame(animate);
    };
    
    // 停止
    const stop = () => {
      isPlaying.value = false;
      currentIndex.value = 0;
      if (animationId) {
        cancelAnimationFrame(animationId);
        animationId = null;
      }
    };
    
    // 上一帧
    const stepBackward = () => {
      if (currentIndex.value > 0) {
        currentIndex.value--;
      }
    };
    
    // 下一帧
    const stepForward = () => {
      if (currentIndex.value < timeSteps.value.length - 1) {
        currentIndex.value++;
      }
    };
    
    // 滑块变化
    const onSliderChange = () => {
      // 加载对应时间的数据
      weatherStore.loadWeatherData(timeSteps.value[currentIndex.value]);
    };
    
    // 监听当前索引变化
    watch(currentIndex, (newIndex) => {
      const time = timeSteps.value[newIndex];
      weatherStore.loadWeatherData(time);
    });
    
    // 初始化
    loadTimeline();
    
    return {
      isPlaying,
      currentIndex,
      playSpeed,
      timeSteps,
      currentTime,
      startTime,
      endTime,
      playPause,
      stop,
      stepBackward,
      stepForward,
      onSliderChange
    };
  }
};
</script>
```

### 3.1 航线渲染器

```javascript
// utils/routeRenderer.js
import * as Cesium from 'cesium';

export class RouteRenderer {
  constructor(viewer) {
    this.viewer = viewer;
    this.routes = new Map();
  }
  
  // 添加航线
  addRoute(routeData) {
    const { id, waypoints } = routeData;
    
    // 创建航线实体
    const positions = waypoints.map(wp => 
      Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat)
    );
    
    const routeEntity = this.viewer.entities.add({
      id: `route-${id}`,
      polyline: {
        positions: positions,
        width: 3,
        material: new Cesium.PolylineGlowMaterialProperty({
          glowPower: 0.2,
          color: Cesium.Color.YELLOW
        })
      }
    });
    
    // 添加航点标记
    waypoints.forEach((wp, index) => {
      this.viewer.entities.add({
        id: `waypoint-${id}-${index}`,
        position: Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat),
        point: {
          pixelSize: 8,
          color: Cesium.Color.YELLOW,
          outlineColor: Cesium.Color.BLACK,
          outlineWidth: 2
        },
        label: {
          text: `WP${index + 1}`,
          font: '12px sans-serif',
          fillColor: Cesium.Color.WHITE,
          outlineColor: Cesium.Color.BLACK,
          outlineWidth: 2,
          pixelOffset: new Cesium.Cartesian2(0, -15)
        }
      });
    });
    
    this.routes.set(id, routeEntity);
    
    return routeEntity;
  }
  
  // 移除航线
  removeRoute(id) {
    const entity = this.routes.get(id);
    if (entity) {
      this.viewer.entities.remove(entity);
      this.routes.delete(id);
    }
  }
  
  // 显示航线气象
  showRouteWeather(routeId, weatherData) {
    weatherData.waypoints.forEach((wp, index) => {
      // 添加气象信息标签
      this.viewer.entities.add({
        id: `weather-${routeId}-${index}`,
        position: Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat, 50000),
        billboard: {
          image: this.createWeatherIcon(wp.weather),
          width: 32,
          height: 32
        },
        label: {
          text: `风速: ${wp.weather.wind.speed.toFixed(1)} m/s\n` +
                `波高: ${wp.weather.wave.height.toFixed(1)} m`,
          font: '10px sans-serif',
          fillColor: Cesium.Color.WHITE,
          backgroundColor: Cesium.Color.BLACK.withAlpha(0.7),
          showBackground: true,
          pixelOffset: new Cesium.Cartesian2(0, 20)
        }
      });
    });
  }
  
  // 创建气象图标
  createWeatherIcon(weather) {
    const canvas = document.createElement('canvas');
    canvas.width = 32;
    canvas.height = 32;
    const ctx = canvas.getContext('2d');
    
    // 根据风速绘制不同颜色
    const speed = weather.wind.speed;
    let color;
    if (speed < 5) color = '#00ff00';
    else if (speed < 10) color = '#ffff00';
    else if (speed < 15) color = '#ff9900';
    else color = '#ff0000';
    
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.arc(16, 16, 12, 0, Math.PI * 2);
    ctx.fill();
    
    // 绘制风向箭头
    const angle = weather.wind.direction * Math.PI / 180;
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(16, 16);
    ctx.lineTo(16 + Math.sin(angle) * 10, 16 - Math.cos(angle) * 10);
    ctx.stroke();
    
    return canvas;
  }
}
```
