## 三、核心代码实现

### 3.1 气象服务 (weatherService.js)

```javascript
const fs = require('fs');
const path = require('path');
const zlib = require('zlib');
const { exec } = require('child_process');
const axios = require('axios');

class WeatherService {
  // 下载 GFS 数据
  static async downloadGFS(date, hour) {
    const dateStr = date.replace(/-/g, '');
    const url = `https://nomads.ncep.noaa.gov/cgi-bin/filter_gfs_0p25.pl?` +
                `file=gfs.t${hour}z.pgrb2.0p25.f000&` +
                `lev_10_m_above_ground=on&` +
                `var_UGRD=on&var_VGRD=on&` +
                `dir=%2Fgfs.${dateStr}%2F${hour}%2Fatmos`;
    
    const response = await axios.get(url, {
      responseType: 'arraybuffer'
    });
    
    const filename = `gfs_${dateStr}_${hour}.grib2`;
    fs.writeFileSync(filename, response.data);
    
    return filename;
  }
  
  // 转换 GRIB2 到 JSON
  static async convertGRIB2(gribFile) {
    return new Promise((resolve, reject) => {
      const jsonFile = gribFile.replace('.grib2', '.json');
      
      exec(`grib2json -d -n -o ${jsonFile} ${gribFile}`, (error) => {
        if (error) {
          reject(error);
        } else {
          resolve(jsonFile);
        }
      });
    });
  }
  
  // 解析风场数据
  static parseWindData(jsonFile) {
    const raw = JSON.parse(fs.readFileSync(jsonFile));
    
    // 找到 U 和 V 分量
    const uData = raw.find(r => 
      r.header.parameterCategory === 2 && 
      r.header.parameterNumber === 2
    );
    const vData = raw.find(r => 
      r.header.parameterCategory === 2 && 
      r.header.parameterNumber === 3
    );
    
    if (!uData || !vData) {
      throw new Error('无法找到风场数据');
    }
    
    const result = {
      refTime: uData.header.refTime,
      bounds: {
        west: uData.header.lo1,
        south: uData.header.la2,
        east: uData.header.lo2,
        north: uData.header.la1
      },
      resolution: uData.header.dx,
      nx: uData.header.nx,
      ny: uData.header.ny,
      data: []
    };
    
    // 合并 U 和 V
    for (let i = 0; i < uData.data.length; i++) {
      result.data.push({
        u: uData.data[i],
        v: vData.data[i]
      });
    }
    
    return result;
  }
  
  // 保存气象数据
  static async saveWeatherData(type, data) {
    const time = new Date(data.refTime).toISOString().replace(/:/g, '-');
    const dir = path.join(__dirname, '../../data/weather', type);
    
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
    }
    
    const filename = path.join(dir, `${time}.json`);
    const json = JSON.stringify(data);
    
    // 压缩
    const compressed = zlib.gzipSync(json);
    fs.writeFileSync(filename + '.gz', compressed);
    
    console.log(`✅ 保存气象数据: ${filename}.gz`);
    
    return filename + '.gz';
  }
  
  // 获取气象数据
  static async getWeatherData(type, time, bounds) {
    const timeStr = new Date(time).toISOString().replace(/:/g, '-');
    const filename = path.join(
      __dirname,
      '../../data/weather',
      type,
      `${timeStr}.json.gz`
    );
    
    if (!fs.existsSync(filename)) {
      throw new Error('气象数据不存在');
    }
    
    // 解压
    const compressed = fs.readFileSync(filename);
    const json = zlib.gunzipSync(compressed).toString();
    const data = JSON.parse(json);
    
    // 如果指定了边界，裁剪数据
    if (bounds) {
      return this.cropData(data, bounds);
    }
    
    return data;
  }
  
  // 裁剪数据到指定边界
  static cropData(data, bounds) {
    const { west, south, east, north } = bounds;
    const { resolution, nx, ny } = data;
    
    const x1 = Math.floor((west - data.bounds.west) / resolution);
    const y1 = Math.floor((data.bounds.north - north) / resolution);
    const x2 = Math.ceil((east - data.bounds.west) / resolution);
    const y2 = Math.ceil((data.bounds.north - south) / resolution);
    
    const croppedData = [];
    
    for (let y = y1; y < y2; y++) {
      for (let x = x1; x < x2; x++) {
        const index = y * nx + x;
        if (index >= 0 && index < data.data.length) {
          croppedData.push(data.data[index]);
        }
      }
    }
    
    return {
      ...data,
      bounds: { west, south, east, north },
      nx: x2 - x1,
      ny: y2 - y1,
      data: croppedData
    };
  }
}

module.exports = WeatherService;
```

### 3.2 插值服务 (interpolationService.js)

```javascript
class InterpolationService {
  // 双线性插值
  static bilinearInterpolation(lat, lon, weatherData) {
    const { bounds, resolution, nx, ny, data } = weatherData;
    
    // 计算网格坐标
    const x = (lon - bounds.west) / resolution;
    const y = (bounds.north - lat) / resolution;
    
    // 边界检查
    if (x < 0 || x >= nx - 1 || y < 0 || y >= ny - 1) {
      return null;
    }
    
    const x0 = Math.floor(x);
    const y0 = Math.floor(y);
    const x1 = x0 + 1;
    const y1 = y0 + 1;
    
    // 获取四个角的数据
    const q00 = data[y0 * nx + x0];
    const q10 = data[y0 * nx + x1];
    const q01 = data[y1 * nx + x0];
    const q11 = data[y1 * nx + x1];
    
    // 插值权重
    const wx = x - x0;
    const wy = y - y0;
    
    // 对每个字段进行插值
    const result = {};
    
    for (const key in q00) {
      result[key] = 
        (1 - wx) * (1 - wy) * q00[key] +
        wx * (1 - wy) * q10[key] +
        (1 - wx) * wy * q01[key] +
        wx * wy * q11[key];
    }
    
    return result;
  }
  
  // 获取航线上的气象数据
  static async getWeatherAlongRoute(waypoints, weatherData) {
    const result = [];
    
    for (const waypoint of waypoints) {
      const weather = this.bilinearInterpolation(
        waypoint.lat,
        waypoint.lon,
        weatherData
      );
      
      if (weather) {
        // 计算风速和风向
        const windSpeed = Math.sqrt(weather.u ** 2 + weather.v ** 2);
        const windDirection = (Math.atan2(weather.u, weather.v) * 180 / Math.PI + 360) % 360;
        
        result.push({
          ...waypoint,
          weather: {
            wind: {
              speed: windSpeed,
              direction: windDirection,
              u: weather.u,
              v: weather.v
            }
          }
        });
      }
    }
    
    return result;
  }
}

module.exports = InterpolationService;
```

### 3.3 定时任务 (weatherUpdateJob.js)

```javascript
const cron = require('node-cron');
const WeatherService = require('../services/weatherService');

// 每 6 小时更新一次气象数据
cron.schedule('0 */6 * * *', async () => {
  console.log('🌦️ 开始更新气象数据...');
  
  try {
    const now = new Date();
    const hour = Math.floor(now.getUTCHours() / 6) * 6;
    const dateStr = now.toISOString().split('T')[0];
    
    // 下载 GFS 数据
    console.log('📥 下载 GFS 数据...');
    const gribFile = await WeatherService.downloadGFS(dateStr, hour.toString().padStart(2, '0'));
    
    // 转换为 JSON
    console.log('🔄 转换 GRIB2 到 JSON...');
    const jsonFile = await WeatherService.convertGRIB2(gribFile);
    
    // 解析风场数据
    console.log('📊 解析风场数据...');
    const windData = WeatherService.parseWindData(jsonFile);
    
    // 保存
    console.log('💾 保存数据...');
    await WeatherService.saveWeatherData('wind', windData);
    
    console.log('✅ 气象数据更新完成');
  } catch (error) {
    console.error('❌ 气象数据更新失败:', error);
  }
});
```
