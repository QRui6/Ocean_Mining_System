## 一、数据匹配算法

### 1.1 核心原理

```
问题：如何获取航线上每个点的气象数据？

航线：一系列经纬度点
气象数据：网格数据（0.25° × 0.25°）

解决：空间插值
```

### 1.2 双线性插值算法

```javascript
// utils/interpolation.js

/**
 * 双线性插值 - 获取任意点的气象数据
 * @param {Number} lat - 纬度
 * @param {Number} lon - 经度
 * @param {Object} weatherData - 气象网格数据
 * @returns {Object} - 插值后的气象数据
 */
function bilinearInterpolation(lat, lon, weatherData) {
  const { bounds, resolution, nx, ny, data } = weatherData;
  
  // 1. 计算在网格中的位置
  const x = (lon - bounds.west) / resolution;
  const y = (bounds.north - lat) / resolution;
  
  // 2. 边界检查
  if (x < 0 || x >= nx - 1 || y < 0 || y >= ny - 1) {
    return null;  // 超出范围
  }
  
  // 3. 找到周围 4 个网格点
  const x0 = Math.floor(x);
  const y0 = Math.floor(y);
  const x1 = x0 + 1;
  const y1 = y0 + 1;
  
  // 4. 获取 4 个角的数据
  const q00 = data[y0 * nx + x0];  // 左上
  const q10 = data[y0 * nx + x1];  // 右上
  const q01 = data[y1 * nx + x0];  // 左下
  const q11 = data[y1 * nx + x1];  // 右下
  
  // 5. 计算插值权重
  const wx = x - x0;  // x 方向权重
  const wy = y - y0;  // y 方向权重
  
  // 6. 双线性插值
  const result = {};
  for (const key in q00) {
    result[key] = 
      (1 - wx) * (1 - wy) * q00[key] +  // 左上
      wx * (1 - wy) * q10[key] +        // 右上
      (1 - wx) * wy * q01[key] +        // 左下
      wx * wy * q11[key];               // 右下
  }
  
  return result;
}

/**
 * 获取航线上所有点的气象数据
 * @param {Array} waypoints - 航点列表 [{lat, lon, time}, ...]
 * @param {Object} windData - 风场数据
 * @param {Object} waveData - 波浪数据
 * @returns {Array} - 航线气象数据
 */
function getWeatherAlongRoute(waypoints, windData, waveData) {
  return waypoints.map(waypoint => {
    // 获取该点的风场数据
    const wind = bilinearInterpolation(
      waypoint.lat,
      waypoint.lon,
      windData
    );
    
    // 获取该点的波浪数据
    const wave = bilinearInterpolation(
      waypoint.lat,
      waypoint.lon,
      waveData
    );
    
    // 计算风速和风向
    const windSpeed = wind ? Math.sqrt(wind.u ** 2 + wind.v ** 2) : 0;
    const windDirection = wind ? 
      (Math.atan2(wind.u, wind.v) * 180 / Math.PI + 360) % 360 : 0;
    
    return {
      ...waypoint,
      weather: {
        wind: {
          speed: windSpeed,
          direction: windDirection,
          u: wind?.u || 0,
          v: wind?.v || 0
        },
        wave: {
          height: wave?.height || 0,
          period: wave?.period || 0,
          direction: wave?.direction || 0
        }
      }
    };
  });
}

module.exports = {
  bilinearInterpolation,
  getWeatherAlongRoute
};
```

## 二、后端 API 实现

### 2.1 航线气象 API

```javascript
// routes/routeWeather.js
const express = require('express');
const router = express.Router();
const { getWeatherAlongRoute } = require('../utils/interpolation');

/**
 * 获取航线气象数据
 * GET /api/route/weather
 * Query:
 *   - mmsi: 船舶 MMSI（可选）
 *   - waypoints: 航点 JSON（可选）
 *   - time: 时间点（可选）
 */
router.get('/route/weather', async (req, res) => {
  try {
    let waypoints;
    
    // 方式 1：通过 MMSI 获取航线
    if (req.query.mmsi) {
      const vessel = await getVesselByMMSI(req.query.mmsi);
      waypoints = vessel.route.waypoints;
    }
    // 方式 2：直接提供航点
    else if (req.query.waypoints) {
      waypoints = JSON.parse(req.query.waypoints);
    }
    else {
      return res.status(400).json({
        success: false,
        error: '需要提供 mmsi 或 waypoints'
      });
    }
    
    // 获取气象数据
    const time = req.query.time || new Date().toISOString();
    const windData = await getWindData(time);
    const waveData = await getWaveData(time);
    
    // 匹配航线气象
    const routeWeather = getWeatherAlongRoute(waypoints, windData, waveData);
    
    // 灾害预警
    const warnings = checkWeatherWarnings(routeWeather);
    
    res.json({
      success: true,
      data: {
        waypoints: routeWeather,
        warnings: warnings,
        summary: {
          maxWindSpeed: Math.max(...routeWeather.map(w => w.weather.wind.speed)),
          maxWaveHeight: Math.max(...routeWeather.map(w => w.weather.wave.height)),
          dangerousPoints: warnings.length
        }
      }
    });
    
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.message
    });
  }
});

/**
 * 灾害预警检查
 */
function checkWeatherWarnings(routeWeather) {
  const warnings = [];
  
  routeWeather.forEach((point, index) => {
    const { wind, wave } = point.weather;
    
    // 风速预警
    if (wind.speed > 20) {  // 大于 20 m/s (约 8 级风)
      warnings.push({
        type: 'wind',
        level: wind.speed > 28 ? 'danger' : 'warning',
        position: { lat: point.lat, lon: point.lon },
        time: point.time,
        message: `强风警告：风速 ${wind.speed.toFixed(1)} m/s`,
        value: wind.speed
      });
    }
    
    // 波浪预警
    if (wave.height > 4) {  // 大于 4 米
      warnings.push({
        type: 'wave',
        level: wave.height > 6 ? 'danger' : 'warning',
        position: { lat: point.lat, lon: point.lon },
        time: point.time,
        message: `大浪警告：波高 ${wave.height.toFixed(1)} m`,
        value: wave.height
      });
    }
  });
  
  return warnings;
}

module.exports = router;
```

## 三、前端可视化实现

### 3.1 航线渲染器

```javascript
// utils/routeWeatherRenderer.js
import * as Cesium from 'cesium';

export class RouteWeatherRenderer {
  constructor(viewer) {
    this.viewer = viewer;
    this.routeEntity = null;
    this.waypointEntities = [];
    this.weatherMarkers = [];
  }
  
  /**
   * 渲染航线和气象
   */
  render(routeWeatherData) {
    this.clear();
    
    const { waypoints, warnings } = routeWeatherData;
    
    // 1. 绘制航线
    this.renderRoute(waypoints);
    
    // 2. 绘制航点气象标记
    this.renderWeatherMarkers(waypoints);
    
    // 3. 绘制预警区域
    this.renderWarnings(warnings);
    
    // 4. 飞到航线
    this.flyToRoute(waypoints);
  }
  
  /**
   * 绘制航线
   */
  renderRoute(waypoints) {
    const positions = waypoints.map(wp => 
      Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat)
    );
    
    this.routeEntity = this.viewer.entities.add({
      polyline: {
        positions: positions,
        width: 4,
        material: new Cesium.PolylineGlowMaterialProperty({
          glowPower: 0.2,
          color: Cesium.Color.YELLOW
        }),
        clampToGround: false
      }
    });
  }
  
  /**
   * 绘制气象标记
   */
  renderWeatherMarkers(waypoints) {
    waypoints.forEach((wp, index) => {
      // 每隔 5 个点显示一个标记
      if (index % 5 !== 0) return;
      
      const { wind, wave } = wp.weather;
      
      // 创建气象图标
      const canvas = this.createWeatherIcon(wind, wave);
      
      const entity = this.viewer.entities.add({
        position: Cesium.Cartesian3.fromDegrees(wp.lon, wp.lat, 50000),
        billboard: {
          image: canvas,
          width: 48,
          height: 48,
          verticalOrigin: Cesium.VerticalOrigin.CENTER
        },
        label: {
          text: `风: ${wind.speed.toFixed(1)} m/s\n浪: ${wave.height.toFixed(1)} m`,
          font: '12px sans-serif',
          fillColor: Cesium.Color.WHITE,
          outlineColor: Cesium.Color.BLACK,
          outlineWidth: 2,
          style: Cesium.LabelStyle.FILL_AND_OUTLINE,
          pixelOffset: new Cesium.Cartesian2(0, 30),
          showBackground: true,
          backgroundColor: Cesium.Color.BLACK.withAlpha(0.7)
        }
      });
      
      this.weatherMarkers.push(entity);
    });
  }
  
  /**
   * 创建气象图标
   */
  createWeatherIcon(wind, wave) {
    const canvas = document.createElement('canvas');
    canvas.width = 48;
    canvas.height = 48;
    const ctx = canvas.getContext('2d');
    
    // 根据风速选择颜色
    let color;
    if (wind.speed < 5) color = '#00ff00';      // 绿色：微风
    else if (wind.speed < 10) color = '#ffff00'; // 黄色：和风
    else if (wind.speed < 15) color = '#ff9900'; // 橙色：强风
    else if (wind.speed < 20) color = '#ff6600'; // 深橙：大风
    else color = '#ff0000';                      // 红色：烈风
    
    // 绘制圆形背景
    ctx.fillStyle = color;
    ctx.beginPath();
    ctx.arc(24, 24, 20, 0, Math.PI * 2);
    ctx.fill();
    
    // 绘制风向箭头
    ctx.strokeStyle = '#000';
    ctx.lineWidth = 3;
    ctx.beginPath();
    
    const angle = wind.direction * Math.PI / 180;
    const x1 = 24;
    const y1 = 24;
    const x2 = 24 + Math.sin(angle) * 15;
    const y2 = 24 - Math.cos(angle) * 15;
    
    ctx.moveTo(x1, y1);
    ctx.lineTo(x2, y2);
    
    // 箭头
    const arrowAngle = Math.PI / 6;
    ctx.lineTo(
      x2 - Math.sin(angle - arrowAngle) * 5,
      y2 + Math.cos(angle - arrowAngle) * 5
    );
    ctx.moveTo(x2, y2);
    ctx.lineTo(
      x2 - Math.sin(angle + arrowAngle) * 5,
      y2 + Math.cos(angle + arrowAngle) * 5
    );
    
    ctx.stroke();
    
    // 如果波浪很大，添加波浪符号
    if (wave.height > 3) {
      ctx.strokeStyle = '#0066ff';
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(10, 40);
      ctx.quadraticCurveTo(15, 35, 20, 40);
      ctx.quadraticCurveTo(25, 45, 30, 40);
      ctx.quadraticCurveTo(35, 35, 38, 40);
      ctx.stroke();
    }
    
    return canvas;
  }
  
  /**
   * 绘制预警区域
   */
  renderWarnings(warnings) {
    warnings.forEach(warning => {
      const color = warning.level === 'danger' ? 
        Cesium.Color.RED : Cesium.Color.ORANGE;
      
      // 绘制预警圆圈
      this.viewer.entities.add({
        position: Cesium.Cartesian3.fromDegrees(
          warning.position.lon,
          warning.position.lat,
          100000
        ),
        ellipse: {
          semiMinorAxis: 50000,  // 50 km
          semiMajorAxis: 50000,
          material: color.withAlpha(0.3),
          outline: true,
          outlineColor: color,
          outlineWidth: 2
        }
      });
      
      // 预警标签
      this.viewer.entities.add({
        position: Cesium.Cartesian3.fromDegrees(
          warning.position.lon,
          warning.position.lat,
          150000
        ),
        label: {
          text: `⚠️ ${warning.message}`,
          font: '14px sans-serif',
          fillColor: Cesium.Color.WHITE,
          outlineColor: Cesium.Color.BLACK,
          outlineWidth: 2,
          style: Cesium.LabelStyle.FILL_AND_OUTLINE,
          showBackground: true,
          backgroundColor: color.withAlpha(0.8),
          pixelOffset: new Cesium.Cartesian2(0, 0)
        }
      });
    });
  }
  
  /**
   * 飞到航线
   */
  flyToRoute(waypoints) {
    if (waypoints.length === 0) return;
    
    // 计算航线的边界
    const lats = waypoints.map(wp => wp.lat);
    const lons = waypoints.map(wp => wp.lon);
    
    const minLat = Math.min(...lats);
    const maxLat = Math.max(...lats);
    const minLon = Math.min(...lons);
    const maxLon = Math.max(...lons);
    
    const centerLat = (minLat + maxLat) / 2;
    const centerLon = (minLon + maxLon) / 2;
    
    // 计算合适的高度
    const latRange = maxLat - minLat;
    const lonRange = maxLon - minLon;
    const maxRange = Math.max(latRange, lonRange);
    const height = maxRange * 200000;  // 根据范围调整高度
    
    this.viewer.camera.flyTo({
      destination: Cesium.Cartesian3.fromDegrees(
        centerLon,
        centerLat,
        height
      ),
      duration: 2
    });
  }
  
  /**
   * 清除所有渲染
   */
  clear() {
    if (this.routeEntity) {
      this.viewer.entities.remove(this.routeEntity);
    }
    
    this.weatherMarkers.forEach(entity => {
      this.viewer.entities.remove(entity);
    });
    
    this.weatherMarkers = [];
  }
}
```

```javascript
// utils/routeTimelinePlayer.js
export class RouteTimelinePlayer {
  constructor(viewer, renderer) {
    this.viewer = viewer;
    this.renderer = renderer;
    this.isPlaying = false;
    this.currentIndex = 0;
    this.timeSteps = [];
  }
  
  /**
   * 加载时序数据
   */
  async loadTimeline(mmsi, startTime, endTime) {
    // 获取多个时间点的航线气象
    const response = await fetch(
      `/api/route/weather/timeline?mmsi=${mmsi}&start=${startTime}&end=${endTime}`
    );
    const result = await response.json();
    
    this.timeSteps = result.data;
    this.currentIndex = 0;
  }
  
  /**
   * 播放动画
   */
  play() {
    if (this.isPlaying) return;
    
    this.isPlaying = true;
    this.animate();
  }
  
  /**
   * 暂停
   */
  pause() {
    this.isPlaying = false;
  }
  
  /**
   * 动画循环
   */
  animate() {
    if (!this.isPlaying) return;
    
    // 渲染当前时间点
    const currentData = this.timeSteps[this.currentIndex];
    this.renderer.render(currentData);
    
    // 下一帧
    this.currentIndex++;
    if (this.currentIndex >= this.timeSteps.length) {
      this.currentIndex = 0;  // 循环播放
    }
    
    // 每秒播放一帧
    setTimeout(() => this.animate(), 1000);
  }
}
```
